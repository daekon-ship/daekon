# Vályog-Mester – éles Apache-konfiguráció (a build .htaccess néven másolja ki)
Options -Indexes
DirectoryIndex index.html
AddDefaultCharset UTF-8
AddType font/woff2 .woff2
AddType image/webp .webp
AddType video/mp4 .mp4
ErrorDocument 404 /404.html

<IfModule mod_rewrite.c>
  RewriteEngine On
  # csak HTTPS, www nélkül
  RewriteCond %{HTTPS} !=on [OR]
  RewriteCond %{HTTP_HOST} ^www\. [NC]
  RewriteRule ^ https://valyogmester.hu%{REQUEST_URI} [R=301,L]
  # belső fájlok tiltása
  RewriteRule (^|/)\.(?!well-known/) - [F,L]
  RewriteRule \.(ini|md|py|log|bak|txt~)$ - [F,L]
</IfModule>

<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=31536000"
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=()"
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self' __CSP_SCRIPT_HASH__; style-src 'self' 'unsafe-inline'; img-src 'self' data: blob:; media-src 'self'; font-src 'self'; connect-src 'self'; form-action 'self'; frame-ancestors 'self'; base-uri 'self'; object-src 'none'"
  <FilesMatch "\.(webp|png|svg|woff2|mp4)$">
    Header set Cache-Control "public, max-age=2592000"
  </FilesMatch>
  <FilesMatch "\.(css|js)$">
    Header set Cache-Control "public, max-age=604800"
  </FilesMatch>
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript image/svg+xml application/json text/plain application/xml
</IfModule>

# PHP feltöltési korlátok (mod_php esetén; PHP-FPM-nél a .user.ini érvényes)
<IfModule mod_php.c>
  php_value upload_max_filesize 10M
  php_value post_max_size 48M
  php_value max_file_uploads 10
  php_value max_execution_time 60
</IfModule>
